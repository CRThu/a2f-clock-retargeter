# 【模式 2】命令行带参调用
# 语法: a2f.bat <工程路径.xpr> [最小时钟扇出:默认4]
D:\Projects\constraint_analyzer\a2f.bat D:\Projects\fm577\fm577_verify\fm577_verify_v2\fm577_vivado.xpr 4

# 或者通过原生 Vivado 批处理调用:
vivado -mode batch -notrace -source D:/Projects/constraint_analyzer/auto_run_retarget.tcl -tclargs <工程路径.xpr>

# ------------------------------------------------------------------------------

# 【模式 3】Vivado GUI 内部交互式调用
